# ============================================================================
# Seafile Android R8/ProGuard rules
#
# Goal: meet Google Play's 25% minimum for Optimization, Obfuscation, and
# Shrinking when the uncompressed DEX size exceeds 10 MB.
#
# Principles:
# 1. Do not keep the whole application; let R8 shrink and obfuscate it.
# 2. Keep only entry points required by reflection, serialization, JNI,
#    WebView, Gson, and similar mechanisms.
# 3. Prefer consumer rules shipped with third-party AAR/JAR dependencies.
# 4. AGP writes the mapping file to build/outputs/mapping automatically.
# ============================================================================

# ------------------------------ Basic attributes ------------------------------
# Reflection libraries such as Gson rely on generic signatures
-keepattributes Signature
# Keep runtime annotations (@SerializedName, @JsonAdapter, Room, Retrofit, etc.)
-keepattributes *Annotation*
# Keep inner-class and enclosing-method metadata for Gson/reflection
-keepattributes InnerClasses,EnclosingMethod
# Keep source file and line numbers for readable crash stack traces
-keepattributes SourceFile,LineNumberTable
-renamesourcefileattribute SourceFile

# Allow R8 to change access modifiers for better optimization
-allowaccessmodification

# ------------------------------ General Android rules ------------------------------
# JNI method names must not be shrunk or obfuscated
-keepclasseswithmembernames class * {
    native <methods>;
}

# Methods bound through android:onClick in XML layouts
-keepclassmembers class * extends android.app.Activity {
    public void *(android.view.View);
}

# Custom View setters/getters used by property animation
-keepclassmembers class * extends android.view.View {
    void set*(***);
    *** get*();
}

# Enum values()/valueOf() are called reflectively
-keepclassmembers enum * {
    public static **[] values();
    public static ** valueOf(java.lang.String);
}

# Parcelable CREATOR fields are accessed reflectively
-keepclassmembers class * implements android.os.Parcelable {
    public static final android.os.Parcelable$Creator CREATOR;
}

# Serializable fields and serialization callbacks
-keepclassmembers class * implements java.io.Serializable {
    static final long serialVersionUID;
    private static final java.io.ObjectStreamField[] serialPersistentFields;
    !static !transient <fields>;
    private void writeObject(java.io.ObjectOutputStream);
    private void readObject(java.io.ObjectInputStream);
    java.lang.Object writeReplace();
    java.lang.Object readResolve();
}

# Resource fields
-keepclassmembers class **.R$* {
    public static <fields>;
}

# Methods exposed to JavaScript through WebView
-keepclassmembers class * {
    @android.webkit.JavascriptInterface <methods>;
}

# ------------------------------ Gson ------------------------------
# Field names annotated with @SerializedName are JSON keys; keep the fields
# themselves so Gson can access them, but allow their Java names to be obfuscated.
-keepclassmembers,allowobfuscation class * {
    @com.google.gson.annotations.SerializedName <fields>;
}

# TypeToken reads generic type information through reflection
-keep class com.google.gson.reflect.TypeToken { *; }
-keep class * extends com.google.gson.reflect.TypeToken { *; }

# Application models: fields are annotated with @SerializedName, so their Java
# names may be obfuscated; keep default constructors for Gson.
-keepclassmembers,allowobfuscation class com.seafile.seadroid2.framework.model.** {
    @com.google.gson.annotations.SerializedName <fields>;
    <init>();
}
-keepclassmembers,allowobfuscation class com.seafile.seadroid2.framework.db.entities.** {
    @com.google.gson.annotations.SerializedName <fields>;
    <init>();
}

# @JsonAdapter adapters are instantiated through their no-argument constructor
-keepclassmembers class com.seafile.seadroid2.framework.model.adapter.** {
    public <init>();
}

# AccountInfo is deserialized directly from api2/account/info/ by Retrofit+Gson.
# It is outside framework.model and has no @SerializedName, so keep its fields
# (and field names) explicitly, otherwise R8 removes them and login/profile
# parsing silently returns empty values or fails at runtime.
-keep class com.seafile.seadroid2.account.AccountInfo {
    <fields>;
    <init>();
}

# ------------------------------ ViewModel ------------------------------
# ViewModelProvider instantiates ViewModels through reflection
-keepclassmembers class * extends androidx.lifecycle.ViewModel {
    public <init>(...);
}

# The base classes below resolve the ViewModel type via
# getClass().getGenericSuperclass(). If the generic base class is not kept,
# R8 reduces the subclass signature from BaseXxx<VM> to the raw BaseXxx and
# the cast to ParameterizedType throws ClassCastException on page startup.
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.BaseActivityWithVM
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.BaseMediaSelectorActivity
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.fragment.BaseFragmentWithVM
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.fragment.BaseDialogFragmentWithVM
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.fragment.BaseBottomSheetDialogFragmentWithVM
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.fragment.RequestBottomSheetDialogFragmentWithVM
-keep,allowobfuscation class com.seafile.seadroid2.ui.base.fragment.RequestCustomDialogFragmentWithVM

# ------------------------------ Media ------------------------------
# metadata-extractor does not use reflection; allow unused parsers to be removed
-dontwarn com.drew.**

# JLatexMath uses Class.forName/getMethod/getDeclaredField internally
-keep class org.scilab.forge.jlatexmath.** { *; }

# ------------------------------ Retrofit ------------------------------
# Retrofit resolves the service method return type reflectively and requires
# Call<Foo>/Single<Foo> to stay parameterized. Retrofit's consumer rules do not
# keep retrofit2.Call itself, so R8 reduces Call<Foo> to the raw Call.
-keep,allowobfuscation class retrofit2.Call

# ------------------------------ RxJava ------------------------------
# RxJava 2.2.10 does not ship consumer rules
-dontwarn java.util.concurrent.Flow*
# Retrofit/RxJava adapters resolve generic return types at runtime. Keep the
# generic container classes so signatures like Single<Foo> are not reduced to
# the raw type (which makes Retrofit throw "return type must be parameterized").
-keep,allowobfuscation class io.reactivex.Single
-keep,allowobfuscation class io.reactivex.Observable
-keep,allowobfuscation class io.reactivex.Flowable
-keep,allowobfuscation class io.reactivex.Maybe
-keepclassmembers class rx.internal.util.unsafe.*ArrayQueue*Field* {
    long producerIndex;
    long consumerIndex;
}

# ------------------------------ Event bus ------------------------------
# LiveEventBus does not ship consumer rules; suppress warnings for optional deps
-dontwarn com.jeremyliao.liveeventbus.**

# ------------------------------ Libraries with consumer rules ------------------------------
# utilcode, Retrofit, OkHttp, Glide, Room, WorkManager, Media3, Firebase,
# AndroidX/Material, and JsBridge (no reflection) are not kept as whole packages
# to avoid unnecessary APK growth.

# ------------------------------ Warnings ------------------------------
# Suppress missing-class warnings from optional third-party dependencies
# (for example javax.annotation and java.awt).
-ignorewarnings
